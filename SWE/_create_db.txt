create database Cricket;

use Cricket;

create table player(
	id int AUTO_INCREMENT PRIMARY KEY,
    name varchar(30),
    country varchar(30),
    club_name varchar(30)
);


use Cricket;

CREATE TABLE club(
	id int AUTO_INCREMENT PRIMARY KEY,
    name varchar(30),
    host_country varchar(30),
    tournament_name varchar(30)
);

CREATE TABLE tournament(
	id int AUTO_INCREMENT PRIMARY KEY,
    name varchar(30),
    host_country varchar(30),
    prize_money int
);
